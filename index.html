// A2Z Portal එකේ ඔබේ Login Details const A2Z_EMAIL = "thameeramanoddaya@gmail.com"; const A2Z_PASSWORD = "123456"; // මෙතනට ඔබේ A2Z Password එක ඇතුළත් කරන්න function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); // 1. Google Sheet එකට Data ඇතුළත් කිරීම (Status එක "Pending" ලෙස වැටේ) sheet.appendRow([ new Date(), data.customerName, data.address, data.city, data.district || '', data.phone1, data.phone2 || '', data.productName, data.qty || 1, data.saleAmount, "Pending" ]); // 2. A2Z Portal එකට Order එක Auto-Submit කිරීම var a2zResponse = sendOrderToA2Z(data); return ContentService.createTextOutput(JSON.stringify({"result": "success", "a2z": a2zResponse})) .setMimeType(ContentService.mimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": err.toString()})) .setMimeType(ContentService.mimeType.JSON); } } function sendOrderToA2Z(orderData) { try { // Step 1: A2Z එකට Login වී Session Cookie එක ලබාගැනීම var loginUrl = "https://a2ztraders.lk/index.php/Dash"; var loginPayload = { "email": A2Z_EMAIL, "password": A2Z_PASSWORD }; var loginOptions = { "method": "post", "payload": loginPayload, "followRedirects": true }; var loginResponse = UrlFetchApp.fetch(loginUrl, loginOptions); var headers = loginResponse.getAllHeaders(); var cookie = headers['Set-Cookie']; // Step 2: Customer Shipping Details A2Z Portal එකට Submit කිරීම var orderUrl = "https://a2ztraders.lk/Customer"; var orderPayload = { "c_name": orderData.customerName, "address": orderData.address, "city": orderData.city, "district": orderData.district || "", "phone_one": orderData.phone1, "phone_two": orderData.phone2 || "", "due_date": "2026-08-23", "pay_method": "COD", "product_name": orderData.productName, "qty": orderData.qty || 1, "sale_price": orderData.saleAmount, "fb_page": "Thameera", "pkg_weight": "1", "mark_wb": "1" }; var orderOptions = { "method": "post", "headers": { "Cookie": cookie }, "payload": orderPayload, "followRedirects": true }; var orderSubmit = UrlFetchApp.fetch(orderUrl, orderOptions); if (orderSubmit.getResponseCode() == 200) { return "Successfully Pushed to A2Z"; } else { return "Failed to Push to A2Z"; } } catch (e) { return "A2Z Error: " + e.toString(); } }
